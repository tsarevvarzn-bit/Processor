//AX - x, BX - y, CX - координата x

PUSH 0
POP AX
PUSH 0
POP BX



CALL :draw_circle
DELAY 5
HLT



:draw_circle

    :y_cycle

        PUSH BX
        PUSH 249
        JA :y_end

        PUSH 0
        POP AX

        :x_cycle

            PUSH AX
            PUSH 249
            JA :x_end

            CALL :draw_point

            PUSH AX
            PUSH 1
            ADD
            POP AX

            JUMP :x_cycle

        :x_end

        PUSH BX
        PUSH 1
        ADD
        POP BX

        JUMP :y_cycle

    :y_end
    DRAW

RET

:draw_point

    PUSH BX
    PUSH 250
    MUL
    PUSH AX
    ADD
    POP DX

//    CALL :get_pixel_number
//  DX - номер пикселя, который рисуем

    PUSH AX
    PUSH 125
    SUB

    PUSH AX
    PUSH 125
    SUB

    MUL

    PUSH BX
    PUSH 125
    SUB

    PUSH BX
    PUSH 125
    SUB

    MUL

    ADD

    PUSH 50
//  Квадрат радиуса

    SUB

    POP CX
    PUSH CX
    PUSH CX
    MUL
    SQRT
    SQRT

    POP CX
    PUSH CX
    PUSH CX

//Взяли модуль разности расстояния до центра и радиуса и размножили

    PUSH 20
//  Половина толщины бублика
    SUB


    POP CX
    PUSH CX
    PUSH CX
//Размножили отклонение от R

    PUSH 0
    JB :missed_point

        PUSH -10
        MUL
        PUSH 200
        ADD
        POP [DX]
        RET

    :missed_point

//      Сбрасываем копию отклонения
        POP [DX]
        PUSH 0
        POP [DX]
        RET
